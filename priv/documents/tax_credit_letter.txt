-- The tax credit letter (#918): what it reads, what it says, and where each part sits.
-- It says what CreateTaxCreditLetter's letter says, word for word.
-- cspell:ignore Łukasz Dvořák Nguyễn

document 'Tax credit letter'
about 'A member''s volunteer hours for a year, for the search and rescue volunteer tax credit.'
for member, year

report member
  select name, address
  from members
  where id = :member

report hours
  select primary, secondary, total
  from tax_credit_hours
  where member.id = :member and year = :year

title '{:year} SRVTC {team.legal_name}' by team.legal_name
lock print
keep hours.primary, hours.secondary

-- What the CRA may question a letter without, so no letter is published while one is
-- blank (#995). Needs attention names each from December (#943).
needs team.mailing_address, team.signer, team.signer_title

-- What the letter's verify page shows anyone holding it, beside the team and the date
-- it was certified.
verify shows
  'Name' member.name
  'Year' :year
  'Primary hours' coalesce(hours.primary, 0) as duration
  'Secondary hours' coalesce(hours.secondary, 0) as duration
  'Total hours' coalesce(hours.total, 0) as duration

-- The email the letter goes out with, to the member, with the PDF attached. Questions go
-- to the signer's email, or their phone when they have no email.
email to member
  subject '{:year} SRVTC {team.legal_name}'
  body
    'Attached is your {:year} tax credit letter from {team.legal_name}.'
    space
    'Claim the search and rescue volunteers amount on line 31240 of your tax return. Keep this letter in case the CRA asks for it.'
    space if team.signer_email is not null or team.signer_phone is not null
    'Questions about your hours? Contact {coalesce(team.signer, team.legal_name)} at {coalesce(team.signer_email, team.signer_phone)}.' if team.signer_email is not null or team.signer_phone is not null

-- Letter paper. Lengths are from the page's edges.
places
  letterhead flow across, top 1in, right 1in, height 1in, gap 0.25in
  letter     flow down, top 1in, left 1in, right 1in, size 12pt, leading 14.4pt

fill
  -- The organization's logo, then the team's, the last in the top right corner. A
  -- missing logo is left out.
  letterhead = organization.logo, team.logo

  -- A blank line drops out. `space` is a blank line, and `space 2` two.
  letter =
    team.mailing_address
    space 2
    'To whom it may concern:'
    space
    'Name: {member.name}'
    'Address: {member.address}' if member.address is not null
    space
    '{member.name} volunteered with {team.legal_name} in {:year}. {team.legal_name} is an eligible search and rescue organization under subsection 118.07(1) of the Income Tax Act.'
    space
    'Primary Hours: {coalesce(hours.primary, 0) as duration}'
    'Secondary Hours: {coalesce(hours.secondary, 0) as duration}'
    'Total Hours: {coalesce(hours.total, 0) as duration}'
    space
    "These hours come from {team.legal_name}'s attendance records."
    space
    if team.signer_phone is null and team.signer_email is null
      then 'If you have questions, contact {team.legal_name}.'
      else 'If you have questions, contact me using the details below.'
    space
    'Certified on {today as date_long}.'
    -- The signer signs by typing their name (#995): /s/ and the name, in a script.
    signature team.signer
    coalesce(team.signer, team.legal_name)
    team.signer_title
    team.signer_phone
    team.signer_email

expect with :year = 2025, team.legal_name = 'Coast Range Search and Rescue Society':
  title = '2025 SRVTC Coast Range Search and Rescue Society'

-- Hours print as the letter list shows them.
expect with hours.primary = 210, hours.secondary = 60, hours.total = 270:
  letter contains 'Primary Hours: 3h 30m'
  and letter contains 'Secondary Hours: 1 hour'
  and letter contains 'Total Hours: 4h 30m'

expect with hours.primary = 45, hours.secondary = 1204, hours.total = 1249:
  letter contains 'Primary Hours: 45 min'
  and letter contains 'Secondary Hours: 20h 4m'
  and letter contains 'Total Hours: 20h 49m'

-- A member with no hours that year gets a letter that says 0 hours.
expect with :year = 2025:
  letter contains 'Total Hours: 0 hours'

expect with member.name = 'Łukasz Dvořák', member.address = '1 Elm St, Hope BC':
  letter contains 'Name: Łukasz Dvořák' and letter contains 'Address: 1 Elm St, Hope BC'

expect with member.name = 'Nguyễn Thị Lan', member.address = null:
  letter contains 'Name: Nguyễn Thị Lan' and letter not contains 'Address:'

expect with team.legal_name = 'Coast Range Search and Rescue Society', :year = 2024,
    member.name = 'Avery Chen':
  letter contains 'Avery Chen volunteered with Coast Range Search and Rescue Society in 2024.'
  and letter contains 'from Coast Range Search and Rescue Society''s attendance records.'

expect with team.legal_name = 'Coast Range Search and Rescue Society':
  letter contains 'If you have questions, contact Coast Range Search and Rescue Society.'

expect with team.signer_email = 'president@example.com':
  letter contains 'If you have questions, contact me using the details below.'

expect with today = date '2026-01-15':
  letter contains 'Certified on January 15, 2026.'

-- The verify code prints the letter's ID and link, so the letter's text ends with
-- the signer.
expect with team.signer_email = 'president@example.com':
  letter ends with 'president@example.com'

-- The signer signs by name, /s/ in front, above their name and title (#995).
expect with team.signer = 'Alex Chen', team.signer_title = 'President':
  letter contains '/s/ Alex Chen' and letter ends with 'President'

-- Without a signer the letter isn't published (needs), and a draft shows no signature.
expect with team.legal_name = 'Coast Range Search and Rescue Society':
  letter not contains '/s/'
