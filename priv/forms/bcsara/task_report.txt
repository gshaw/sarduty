-- EMBC's Task Report Form, for BCSARA's teams (#950, #993): the page D4H draws, which
-- EMCR takes today, not the 2013 page on EMCR's website. It prints on the blank page
-- beside it, task_report.pdf: Andrew's version 4 with every value taken out.
--
-- The page's blanks are drawn rules, not underscores, so tools/forms/find_places.py
-- can't measure them, and each comment below says "on the line" so the script leaves
-- the place alone. Each place was measured from the rule under it (pdfplumber, points
-- from the page's top left): the value's baseline sits 1.5pt above the rule, 2pt in from
-- each end. The two boxes keep the original's 8.6pt inset.
--
-- The task leader signs by publishing it, or a team admin signs in their own name for
-- them (#993): "/s/ Cameron Ruiz", or "/s/ Alex Chen for Cameron Ruiz", on the Task
-- Leader's Signature line, and the day it was signed on the Date beside it. A draft is
-- never signed.
--
-- Details of Task and Equipment Used/Lost shrink to 8pt, then go on an attached page
-- with the whole text (#993), rather than below 8pt or past their box.
--
-- Left blank for the people who sign by hand: EPC Name, its Signature and Date, and all
-- of section B.

form 'EMBC Task Report Form'
about 'The EMBC task report after an incident, signed by its task leader.'
for activity where kind = 'incident'

report task
  select ref_id, tracking_number, `Agency`, `Agency File #`, `ASE#`, `Region`, `Task Type`,
    started_at, finished_at, attended_count, attended_hours, description,
    `Equipment Used/Lost`, `Task Leader's Name`, finished
  from activities
  where id = :activity

compute
  total_person_hours = round(hours(task.attended_hours), 1)

title 'Task report {coalesce(task.tracking_number, task.ref_id)}' by team.name
keep task.tracking_number, task.`Agency File #`, task.attended_count, task.attended_hours

-- What EMCR can't take the form without, from Andrew's filled-in page (#993), so it
-- never publishes with one blank. Task Completed must be a time that has passed. ASE #,
-- the EPC lines, and section B stay optional.
needs task.tracking_number, task.`Agency`, task.`Agency File #`, task.`Region`,
  task.`Task Type`, task.description, task.`Task Leader's Name`, task.finished

-- The task leader signs it: the one current member with that name (#993).
signed by task.`Task Leader's Name`

-- No verify code on EMBC's form for now: Andrew asked that it stay a drop-in
-- replacement for the current process (#970). Each PDF still gets an ID.
verify code none

-- What its verify page shows anyone holding it: never a name.
verify shows
  'Task number' coalesce(task.tracking_number, task.ref_id)
  'SAR person hours' task.attended_hours as hours

page

places
  task_number  text top 106.4pt, left 88.8pt, width 110.0pt, fit, align center, size 9pt  -- on the line after 'Task:'
  agency       text top 106.4pt, left 273.6pt, width 64.4pt, fit, align center, size 9pt  -- on the line after 'Agency:'
  file_number  text top 106.4pt, left 412.8pt, width 64.4pt, fit, align center, size 9pt  -- on the line after 'File Number:'
  ase          text top 106.4pt, left 552.0pt, width 41.6pt, fit, align center, size 9pt  -- on the line after 'ASE #:'
  group        text top 164.6pt, left 112.2pt, width 191.6pt, fit, align center, size 9pt  -- on the line after 'PSLV Group:'
  region       text top 164.6pt, left 402.0pt, width 191.6pt, fit, align center, size 9pt  -- on the line after 'Region:'
  task_type    text top 183.8pt, left 112.8pt, width 480.8pt, fit, align center, size 9pt  -- on the line after 'Task Type:'
  commenced    text top 203.0pt, left 112.2pt, width 191.6pt, fit, align center, size 9pt  -- on the line after 'Task Commenced:'
  completed    text top 203.0pt, left 402.0pt, width 191.6pt, fit, align center, size 9pt  -- on the line after 'Task Completed:'
  volunteers   text top 232.4pt, left 112.2pt, width 191.6pt, fit, align center, size 9pt  -- on the line after 'Number of PSLV Volunteers:'
  person_hours text top 232.4pt, left 402.0pt, width 191.6pt, fit, align center, size 9pt  -- on the line after 'Number of Person Hours:'
  details      text top 263.0pt, left 125.7pt, width 461.0pt, pitch 12pt, lines 10, size 9pt, more 'Details of Task'  -- in the box after 'Details of Task:'
  equipment    text top 399.5pt, left 125.7pt, width 461.0pt, pitch 12pt, lines 4, size 9pt, more 'Equipment Used/Lost'  -- in the box after 'Equipment Used/Lost:'
  leader       text top 460.4pt, left 112.8pt, width 480.8pt, fit, align center, size 9pt  -- on the line after 'Task Leader's Name:'
  signed       signature top 479.6pt, left 112.2pt, width 191.6pt, size 9pt  -- on the line after the first 'Signature:'
  signed_on    text top 479.6pt, left 402.0pt, width 191.6pt, fit, align center, size 9pt  -- on the line after the first 'Date:'

fill
  task_number = task.tracking_number
  agency = task.`Agency`
  file_number = task.`Agency File #`
  ase = task.`ASE#`
  group = team.name
  region = task.`Region`
  task_type = task.`Task Type`
  commenced = '{task.started_at as date_short}, {task.started_at as time}' if task.started_at is not null
  completed = '{task.finished_at as date_short}, {task.finished_at as time}' if task.finished_at is not null
  volunteers = task.attended_count as number
  person_hours = total_person_hours
  details = task.description
  equipment = task.`Equipment Used/Lost`
  leader = task.`Task Leader's Name`
  signed = signer.signature
  signed_on = signer.signed_on as date_short

expect with task.tracking_number = '26-06613', task.`Agency` = 'Police',
  task.`Agency File #` = '26-88481', task.`Task Type` = '002 - Land SAR':
  task_number = '26-06613' and agency = 'Police' and file_number = '26-88481'
  and task_type = '002 - Land SAR' and title = 'Task report 26-06613'

-- With no task number yet, the PDF is named for the activity's ID.
expect with task.ref_id = '01428', task.tracking_number = '': title = 'Task report 01428'

-- Person hours to one decimal, as EMBC's form shows them: 9,858 minutes is 164.3 hours.
expect with task.attended_count = 22, task.attended_hours = 9858:
  volunteers = '22' and person_hours = '164.3'

-- The task leader signs by publishing, dated the day they do (#993).
expect with signer.name = 'Cameron Ruiz', signer.signed_on = 2026-11-11:
  signed = '/s/ Cameron Ruiz' and signed_on = 'Nov 11, 2026'

-- A team admin signs in their own name, for the task leader.
expect with signer.name = 'Alex Chen', signer.signed_for = 'Cameron Ruiz':
  signed = '/s/ Alex Chen for Cameron Ruiz'

-- A draft is never signed.
expect with task.`Task Leader's Name` = 'Cameron Ruiz': signed is null and signed_on is null
